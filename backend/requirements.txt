# dependency set. Installing it produced a backend that cannot
# boot (no sqlalchemy / celery / redis / flask-limiter / ...).

# locked by uv.lock. Install with:
#
#     uv sync --frozen --group dev
#

# does not match fails the build. Adding a dependency means
# editing pyproject.toml and re-locking — never editing this
# file. This file remains only so that `pip install -r
# requirements.txt` is answered by the pointer instead of
# silently installing a broken environment.
